# 0015 — 3Sum

## Deskripsi

**Soal:** [0015. 3Sum](https://leetcode.com/problems/3sum/)

**Tingkat Kesulitan:** Sedang

**Kompleksitas Waktu:** O(n^2)  
**Kompleksitas Ruang:** O(1) (excluding output)

**Algoritma:** —

**Fungsi Solusi:** `func threeSum(nums []int) [][]int`

## Solusi Go

```go
package main

// LeetCode #15: 3Sum
// https://leetcode.com/problems/3sum/
// Difficulty: Medium

import (
	"fmt"
	"sort"
)

func threeSum(nums []int) [][]int {
	sort.Ints(nums)
	result := [][]int{}

  // Loop standar: indeks 0 sampai n-1
	for i := 0; i < len(nums)-2; i++ {
		if i > 0 && nums[i] == nums[i-1] {
			continue
		}
		left, right := i+1, len(nums)-1
  // Loop two-pointer: kiri vs kanan
		for left < right {
			sum := nums[i] + nums[left] + nums[right]
			if sum == 0 {
				result = append(result, []int{nums[i], nums[left], nums[right]})
  // Loop two-pointer: kiri vs kanan
				for left < right && nums[left] == nums[left+1] {
					left++
				}
  // Loop two-pointer: kiri vs kanan
				for left < right && nums[right] == nums[right-1] {
					right--
				}
				left++
				right--
			} else if sum < 0 {
				left++
			} else {
				right--
			}
		}
	}

	return result
}

func main() {
	// Test case 1
	fmt.Println(threeSum([]int{-1, 0, 1, 2, -1, -4})) // [[-1 -1 2] [-1 0 1]]

	// Test case 2
	fmt.Println(threeSum([]int{0, 1, 1})) // []

	// Test case 3
	fmt.Println(threeSum([]int{0, 0, 0})) // [[0 0 0]]
}

// Time: O(n^2) | Space: O(1) (excluding output)
```
